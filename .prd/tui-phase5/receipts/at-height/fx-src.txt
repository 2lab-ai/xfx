   99  /// Paces the frame for a keystroke that started a folder listing. Most folders
  100  /// list well within one event-loop tick, so that frame waits for the rows and
  101  /// the keystroke paints together with them instead of over an empty
  102  /// "indexing files..." list. Times are monotonic milliseconds. Modeled in
  103  /// `tests/tla/indexer/ListingWait.tla`.
  104  const FileListingWait = struct {
  105      const hold_ms: i64 = 8;
  106  
  107      started_ms: ?i64 = null,
  108      // The previous listing outlasted the wait, so the next keystroke paints at
  109      // once instead of paying the wait again on a slow folder.
  110      last_outlasted_hold: bool = false,
  111  
  112      /// Returns true while the next frame should wait for an in-flight listing.
  113      /// Keystrokes during one listing share its wait.
  114      pub fn hold(self: *FileListingWait, listing_in_flight: bool, now_ms: i64) bool {
  115          if (!listing_in_flight) {
  116              if (self.started_ms) |started| self.last_outlasted_hold = !withinHold(started, now_ms);
  117              self.started_ms = null;
  118              return false;
  119          }
  120          if (self.started_ms == null) self.started_ms = now_ms;
  121          return self.active(now_ms);
  122      }
  123  
  124      pub fn active(self: FileListingWait, now_ms: i64) bool {
  125          return !self.last_outlasted_hold and self.measuring(now_ms);
  126      }
  127  
  128      /// True during the first tick of every listing, waited for or not, so a
  129      /// listing that finishes in time is seen at once and waiting turns back on.
  130      pub fn measuring(self: FileListingWait, now_ms: i64) bool {
  131          const started = self.started_ms orelse return false;
  132          return withinHold(started, now_ms);
  133      }
  134  
  135      fn withinHold(started_ms: i64, now_ms: i64) bool {
  136          // A clock that steps backwards ends the wait instead of extending it.
  137          return now_ms >= started_ms and now_ms - started_ms < hold_ms;
  138      }
