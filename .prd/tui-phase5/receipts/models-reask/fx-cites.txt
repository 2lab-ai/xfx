   28  /// session never polls the models endpoint.
   29  pub const LoadTrigger = enum {
   30      /// Refetches a loaded catalog only after the provider's refresh interval.
   31      activity,
   32      /// The model picker opened. Revalidates unless a fetch is in flight or
   33      /// one started less than `picker_revalidate_floor_ms` ago.
   34      picker_open,
   35  };
   36  
   37  /// Reopening the picker within this window reuses the catalog it just fetched.
   38  const picker_revalidate_floor_ms: i64 = 5_000;
   39  
   40  /// After consecutive failures no trigger fetches again for 5 s, then 10 s,
   41  /// 20 s, and so on, up to five minutes, so an outage is never hammered.
   42  const failure_backoff_base_ms: i64 = 5_000;
   43  const failure_backoff_cap_ms: i64 = 5 * std.time.ms_per_min;
   44  
   45  /// How long a request waits for a failed catalog's retry before it goes out
   46  /// without the catalog's limits.
   47  const failed_catalog_retry_wait_ms: i64 = 2_000;
   48  
   49  fn failureBackoffMs(failure_streak: u32) i64 {
   50      var delay = failure_backoff_base_ms;
   51      var step: u32 = 1;
   52      while (step < failure_streak and delay < failure_backoff_cap_ms) : (step += 1) delay *= 2;
   53      return @min(delay, failure_backoff_cap_ms);
   54  }

  582                  !std.meta.eql(self.requested_access.?, requested_access));

  584          // A fetch started or staged under the old access would overwrite the
  585          // reused provenance, so only an idle ready catalog is reused; anything
  586          // else resets below.
  587          if (access_changed and
  588              self.state == .ready and
  589              !self.fetch_in_flight and
  590              self.staged == null and
  591              requested_access.level == .public_only)
  592          {
  593              if (self.outcome.loaded) |*loaded| {
  594                  if (loaded.access.level == .public_only) {
  595                      self.requested_access = requested_access;
  596                      loaded.* = .{ .access = requested_access };
  597                      self.outcome.last_failure = null;
  598                      if (self.menu.active) {
  599                          self.menu.catalog_state = modelMenuCatalogState(self.outcome, self.from_profile_settings);
  600                      }
  601                      reused_public_catalog = true;
  602                  }
  603              }
  604          }
  605          self.mutex.unlock(io_mod.getIo());
  606          if (reused_public_catalog) return false;
  607          if (access_changed) self.reset();
  608  
  609          const now = model_catalog.clockMs();
  610          self.mutex.lockUncancelable(io_mod.getIo());
  611          // A clock that moved backwards makes every elapsed check due.
  612          const elapsed = if (now < self.last_attempt_ms) std.math.maxInt(i64) else now - self.last_attempt_ms;
  613          const expired = if (refresh_interval_ms) |interval| elapsed >= interval else false;
  614          const picker_due = trigger == .picker_open and elapsed >= picker_revalidate_floor_ms;
  615          // A retained catalog whose refresh failed retries once its backoff ends.
  616          const retry_due = if (self.outcome.last_failure) |failed| failed.failure.retryable else false;
  617          const backoff_over = now < self.last_attempt_ms or now >= self.retry_not_before_ms;
  618          // A staged revalidation is newer than anything a fetch would start now.
  619          // A fetch that finished after the join above still holds its thread;
  620          // the next trigger joins it before starting another.
  621          const should_load = !self.fetch_in_flight and self.staged == null and self.thread == null and backoff_over and switch (self.state) {
  622              .idle, .failed => true,
  623              .ready => expired or picker_due or retry_due,
  624              .loading => false,
  625          };
  626          if (!should_load) {
  627              self.mutex.unlock(io_mod.getIo());
  628              return false;
  629          }
  630          // A first load blocks request resolution, and a failed catalog's retry
  631          // holds it for at most `failed_catalog_retry_wait_ms`. A revalidation
  632          // runs behind the state readers already see.
  633          const first_load = self.state == .idle;
  634          if (first_load) self.state = .loading;
  635          self.fetch_in_flight = true;
  636          self.last_attempt_ms = now;
  637          self.requested_access = requested_access;
  638          const failure_streak = self.failure_streak;
  639          self.mutex.unlock(io_mod.getIo());
  640          self.cancel_requested.store(false, .seq_cst);

  936      fn completeFetch(self: *Self, result: model_catalog.FetchResult) void {
  937          self.mutex.lockUncancelable(io_mod.getIo());
  938          defer self.mutex.unlock(io_mod.getIo());
  939          self.fetch_in_flight = false;
  940          self.completion_pending = true;
  941          switch (result) {
  942              .loaded => {
  943                  self.failure_streak = 0;
  944                  self.retry_not_before_ms = 0;
  945              },
  946              // Cancellation comes only from a reset or shutdown, which
  947              // discards this result, so it is not a failure to back off from.
  948              .failed => |failed| if (failed.failure.category != .cancellation) {
  949                  self.failure_streak +|= 1;
  950                  const delay_ms = failureBackoffMs(self.failure_streak);
  951                  self.retry_not_before_ms = model_catalog.clockMs() + delay_ms;
  952                  debug_trace.logf("catalog", "model catalog fetch failed category={s} failure_streak={d} retry_in_ms={d}", .{
  953                      @tagName(failed.failure.category),
  954                      self.failure_streak,
  955                      delay_ms,
  956                  });
  957              },
  958          }
  959          if (self.state == .ready) {
  960              std.debug.assert(self.staged == null);
  961              self.staged = result;
  962              return;
  963          }
  964          self.applyFetchLocked(result).record();
  965      }

  967      /// Installs a fetch result, keeping the previous catalog when the new one
  968      /// failed or came back empty. Takes ownership of a loaded catalog.
  969      fn applyFetchLocked(self: *Self, result: model_catalog.FetchResult) AppliedFetch {
  970          switch (result) {
  971              .failed => |failure| {
  972                  self.outcome.last_failure = failure;
  973                  self.state = if (self.outcome.loaded != null and self.catalog.items.len > 0) .ready else .failed;
  974                  return .{ .failed = .{ .outcome = failure, .kept_previous = self.state == .ready } };
  975              },
  976              .loaded => |loaded| {
  977                  var catalog = loaded.catalog;
  978                  const reuse_previous = catalog.items.len == 0 and self.outcome.loaded != null and self.catalog.items.len > 0;
  979                  if (reuse_previous) {
  980                      model_catalog.freeModelCatalog(self.alloc, &catalog);
  981                      self.outcome.last_failure = if (loaded.provenance.fallback_failure) |failure| .{
  982                          .access = loaded.provenance.access,
  983                          .anonymous_fallback_used = loaded.provenance.anonymous_fallback_used,
  984                          .failure = failure,
  985                      } else null;
  986                  } else {
  987                      model_catalog.freeModelCatalog(self.alloc, &self.catalog);
  988                      self.catalog = catalog;
  989                      self.outcome = .{ .loaded = loaded.provenance };
  990                  }
  991                  self.state = .ready;
  992                  return .{ .ready = .{
  993                      .entries = self.catalog.items.len,
  994                      .reused_previous = reuse_previous,
  995                      .provenance = loaded.provenance,
  996                  } };
  997              },
  998          }
  999      }
