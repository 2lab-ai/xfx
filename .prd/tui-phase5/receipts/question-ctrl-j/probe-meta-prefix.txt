        // A probe, not committed: `ESC ESC [106;5u` (fx's meta-prefixed
        // Ctrl+J) at a choice and in the field, on xfx's decoder.
        for field in [false, true] {
            let mut fixture = shell(24, 120);
            let started = asking_question(&mut fixture, a_batch());
            if field {
                fixture.route_bytes(b"3");
                fixture.route_bytes(b"ab");
            }
            fixture.route_bytes(b"\x1b\x1b[106;5u");
            fixture.settle_input(started + Decoder::ESC_TIMEOUT * 4);
            let rows = fixture.band_rows();
            let shown = ["Which depth?", "Ship it?"]
                .into_iter()
                .find(|title| rows.iter().any(|row| row.contains(title)));
            let sent = fixture.controlled();
            let hint = fixture.hint();
            let composer = fixture.editor.text().to_string();
            println!(
                "PROBE field {field}: control channel {:?}; question up {}; shown {:?}; composer {:?}; hint {:?}",
                sent,
                fixture.ask.is_some(),
                shown,
                composer,
                hint
            );
        }
    }
