   76  pub fn run(terminal: anytype, should_exit: *bool, poll_timeout_ms: i32, callbacks: EventLoopCallbacks) !ExitCause {
   77      var buf: [128]u8 = undefined;
   78  
   79      while (!should_exit.*) {
   80          try callbacks.collect_facts(callbacks.ctx);
   81          if (should_exit.*) return .requested_exit;
   82  
   83          while (callbacks.next_collected_byte(callbacks.ctx)) |byte| {
   84              try callbacks.handle_byte(callbacks.ctx, byte);
   85              if (should_exit.*) return .requested_exit;
   86          }
   87  
   88          const timeout_ms = if (callbacks.poll_timeout_ms) |resolve|
   89              resolve(callbacks.ctx, poll_timeout_ms)
   90          else
   91              poll_timeout_ms;
   92          var poll_result = try terminal.pollInput(timeout_ms);
   93          var input_reads: usize = 0;
   94          while (poll_result.readable and input_reads < max_input_reads_per_fact_collection) : (input_reads += 1) {
   95              const read_len = try terminal.read(&buf);
   96              if (read_len == 0) {
   97                  if (input_reads > 0) attemptFinalFrameCommit(callbacks, "eof");
   98                  return .input_closed;
   99              }
  100  
  101              record_tape.recordStdin(buf[0..read_len]);
  102              for (buf[0..read_len]) |byte| {
  103                  try callbacks.handle_byte(callbacks.ctx, byte);
  104                  if (should_exit.*) return .requested_exit;
  105              }
  106              poll_result = try terminal.pollInput(0);
  107          }
  108  
  109          if (poll_result.readable) {
  110              try callbacks.commit_frame(callbacks.ctx);
  111              continue;
  112          }
  113          if (poll_result.closed()) {
  114              if (input_reads > 0) attemptFinalFrameCommit(callbacks, "hangup");
  115              return .input_closed;
  116          }
  117  
  118          try callbacks.settle_delivery_epoch(callbacks.ctx);
  119          try callbacks.commit_frame(callbacks.ctx);
  120      }
  121      return .requested_exit;
  122  }
