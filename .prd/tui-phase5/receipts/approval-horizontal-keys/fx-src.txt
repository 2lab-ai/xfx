  185  fn kittyUnicodeKeyAction(keycode: u16, modifiers: u16, meta_prefixed: bool) InputEscapeAction {
  186      // Strip Caps Lock (bit 6) and Num Lock (bit 7) — lock states, not modifiers.
  187      const mods = modifiers & 0x3F;
  188      if (keycode == 27 and mods == 0) return .escape;
  189      // Hyper and Meta survive the lock-state mask, so a keypad key held with one
  190      // of them keeps today's behavior instead of typing text or moving the caret.
  191      if ((mods & ~keypad_supported_modifiers) == 0) {
  192          if (keypadCharacterKey(keycode)) |byte| {
  193              // Only a bare or Shift-modified keypad press types text, matching
  194              // the main row where modified digits and operators are not text.
  195              if (!meta_prefixed and (mods & ~shift_modifier) == 0) return .{ .remapped_byte = byte };
  196          } else if (keypadNavigationKey(keycode)) |letter| {
  197              return navigationKeyAction(letter, mods, meta_prefixed);
  198          } else if (keycode == kp_page_up or keycode == kp_page_down) {
  199              return pageKeyAction(keycode, mods);
  200          } else if (keycode == kp_enter) {
  201              // Reuse the main-row Enter branch: Ctrl submits, Shift and Alt
  202              // insert a newline, and a bare press submits.
  203              return kittyUnicodeKeyAction(13, modifiers, meta_prefixed);
  204          } else if (keycode == kp_delete) {
  205              return forwardDeleteKeyAction(mods);
  206          }
  207      }
  208      if (keycode == kitty_up_key or keycode == kitty_down_key) {
  209          if (meta_prefixed or mods != 0) {
  210              return modifiedArrowAction(
  211                  if (keycode == kitty_up_key) 'A' else 'B',
  212                  mods,
  213                  meta_prefixed,
  214              ) orelse .ignore;
  215          }
  216          return if (keycode == kitty_up_key) .cursor_up else .cursor_down;
  217      }
