# The probe:
    #[test]
    fn zz_sweep_probe() {
        let evil = "P\u{202e}Q\u{200b}R\u{1b}[2JS\u{9b}T\tU";
        // What the channel does to every text a UiEvent carries.
        let inert = |s: &str| bridge::inert(s).into_owned();
        let mut record = record();
        record.welcome();
        record.full_detail(Notice::neutral("recovery", inert(&format!("record {evil}"))));
        record.prompt(&inert(&format!("prompt {evil}")));
        record.answer(&inert(&format!("answer {evil}")));
        record.notice(&Notice::warning("network", inert(&format!("notice {evil}"))));
        record.block(&[inert(&format!("block {evil}"))]);
        record.tool_started("c", &command(&format!("heading {evil}")));
        record.tool_finished("c", true, "", Outcome::Exited(0), None);
        record.usage(&inert(&format!("usage {evil}")));
        record.open();
        for row in record.rows(120, 40, DARK, "") {
            let raw = row.chars().any(|c| matches!(c as u32, 0x200b..=0x200f | 0x2028..=0x202e | 0x80..=0x9f | 0xfeff));
            println!("PROBE raw_mark={raw} esc2J={} {:?}", row.contains("\u{1b}[2J"), unpainted(&row));
        }
    }
