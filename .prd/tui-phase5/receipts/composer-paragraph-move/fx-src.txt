  185  fn kittyUnicodeKeyAction(keycode: u16, modifiers: u16, meta_prefixed: bool) InputEscapeAction {
  186      // Strip Caps Lock (bit 6) and Num Lock (bit 7) — lock states, not modifiers.
  187      const mods = modifiers & 0x3F;
  188      if (keycode == 27 and mods == 0) return .escape;
  189      // Hyper and Meta survive the lock-state mask, so a keypad key held with one
  190      // of them keeps today's behavior instead of typing text or moving the caret.
  191      if ((mods & ~keypad_supported_modifiers) == 0) {
  192          if (keypadCharacterKey(keycode)) |byte| {
  193              // Only a bare or Shift-modified keypad press types text, matching
  194              // the main row where modified digits and operators are not text.
  195              if (!meta_prefixed and (mods & ~shift_modifier) == 0) return .{ .remapped_byte = byte };
  196          } else if (keypadNavigationKey(keycode)) |letter| {
  197              return navigationKeyAction(letter, mods, meta_prefixed);
  198          } else if (keycode == kp_page_up or keycode == kp_page_down) {
  199              return pageKeyAction(keycode, mods);
  200          } else if (keycode == kp_enter) {
  201              // Reuse the main-row Enter branch: Ctrl submits, Shift and Alt
  202              // insert a newline, and a bare press submits.
  203              return kittyUnicodeKeyAction(13, modifiers, meta_prefixed);
  204          } else if (keycode == kp_delete) {
  205              return forwardDeleteKeyAction(mods);
  206          }
  207      }
  208      if (keycode == kitty_up_key or keycode == kitty_down_key) {
  209          if (meta_prefixed or mods != 0) {
  210              return modifiedArrowAction(
  211                  if (keycode == kitty_up_key) 'A' else 'B',
  212                  mods,
  213                  meta_prefixed,
  214              ) orelse .ignore;
  215          }
  216          return if (keycode == kitty_up_key) .cursor_up else .cursor_down;
  217      }
  218      if (keycode == 13 and mods == ctrl_modifier and !meta_prefixed) return .{ .remapped_byte = '\r' };
  219      if (keycode == 13 and (mods & (shift_modifier | alt_modifier)) != 0) {
  220          return .insert_newline;
  221      }
  222      if (keycode == ' ' and mods == shift_modifier and !meta_prefixed) return .{ .remapped_byte = ' ' };
  223      if ((keycode == 'a' or keycode == 'A') and (mods & super_modifier) != 0) {
  224          return .{ .composer_shortcut = .select_all };
  225      }
  226      if ((keycode == 'c' or keycode == 'C') and (mods & super_modifier) != 0) {
  227          return .{ .composer_shortcut = .copy_selection };
  228      }
  229      if ((keycode == 'x' or keycode == 'X') and (mods & super_modifier) != 0) {
  230          return .{ .composer_shortcut = .cut_selection };
  231      }
  232      if ((keycode == 'z' or keycode == 'Z') and (mods & super_modifier) != 0) {
  233          return .{ .composer_shortcut = if ((mods & shift_modifier) != 0) .redo else .undo };
  234      }
  235      if ((mods & (shift_modifier | ctrl_modifier)) ==
  236          (shift_modifier | ctrl_modifier))
  237      {
  238          if (keycode == 'a' or keycode == 'A') return composerMove(.line_start, mods);
  239          if (keycode == 'b' or keycode == 'B') return composerMove(.character_left, mods);
  240          if (keycode == 'e' or keycode == 'E') return composerMove(.line_end, mods);
  241          if (keycode == 'f' or keycode == 'F') return composerMove(.character_right, mods);
  242      }
  243      if (keycode == 'b' and (mods & alt_modifier) != 0) {
  244          if ((mods & shift_modifier) != 0) return composerMove(.word_left, mods);
  245          return .word_left;
  246      }
  247      if (keycode == 'f' and (mods & alt_modifier) != 0) {
  248          if ((mods & shift_modifier) != 0) return composerMove(.word_right, mods);
  249          return .word_right;
  250      }
  251      if ((keycode == 'd' or keycode == 'D') and (meta_prefixed or (mods & 0x02) != 0)) return .delete_word_right;
  252      if (keycode == 'o' or keycode == 'O') return ctrlOKeyAction(meta_prefixed, mods);
  253      if ((keycode == 'r' or keycode == 'R') and (mods & 0x08) != 0) return .open_all_sessions;
  254      if (keycode == 9 and (mods & 0x01) != 0) return .toggle_permission_mode;
  255      if ((mods & 0x04) != 0) {
  256          if (keycode == '_') return .{ .remapped_byte = 31 };
  257          if (keycode >= 'a' and keycode <= 'z') return .{ .remapped_byte = @intCast(keycode - 96) };
  258          if (keycode >= 'A' and keycode <= 'Z') return .{ .remapped_byte = @intCast(keycode - 64) };
  259      }
  260      if (keycode == 127 and (mods & 0x08) != 0) return .delete_to_line_start;
  261      if (keycode == 127 and (mods & 0x02) != 0) return .delete_word_left;
  262      if (keycode == 13) return .{ .remapped_byte = '\r' };
  263      if (keycode == 9) return .{ .remapped_byte = '\t' };
  264      if (keycode == 127) return .{ .remapped_byte = 127 };
  265      return .ignore;

  104  pub fn previousParagraphStart(text: []const u8, cursor: usize) usize {
  105      var line_start = logicalLineStart(text, cursor);
  106      var line_end = logicalLineEnd(text, line_start);
  107  
  108      while (lineIsBlank(text, line_start, line_end)) {
  109          line_start = previousLineStart(text, line_start) orelse return 0;
  110          line_end = logicalLineEnd(text, line_start);
  111      }
  112  
  113      var paragraph_start = line_start;
  114      while (previousLineStart(text, paragraph_start)) |candidate| {
  115          if (lineIsBlank(text, candidate, logicalLineEnd(text, candidate))) break;
  116          paragraph_start = candidate;
  117      }
  118      if (cursor > paragraph_start) return paragraph_start;
  119  
  120      var previous = previousLineStart(text, paragraph_start) orelse return 0;
  121      while (lineIsBlank(text, previous, logicalLineEnd(text, previous))) {
  122          previous = previousLineStart(text, previous) orelse return 0;
  123      }
  124      var previous_start = previous;
  125      while (previousLineStart(text, previous_start)) |candidate| {
  126          if (lineIsBlank(text, candidate, logicalLineEnd(text, candidate))) break;
  127          previous_start = candidate;
  128      }
  129      return previous_start;
  130  }

  132  pub fn nextParagraphStart(text: []const u8, cursor: usize) usize {
  133      var line_start = logicalLineStart(text, cursor);
  134      var line_end = logicalLineEnd(text, line_start);
  135  
  136      while (!lineIsBlank(text, line_start, line_end)) {
  137          if (line_end == text.len) return text.len;
  138          line_start = line_end + 1;
  139          line_end = logicalLineEnd(text, line_start);
  140      }
  141      while (lineIsBlank(text, line_start, line_end)) {
  142          if (line_end == text.len) return text.len;
  143          line_start = line_end + 1;
  144          line_end = logicalLineEnd(text, line_start);
  145      }
  146      return line_start;
  147  }
