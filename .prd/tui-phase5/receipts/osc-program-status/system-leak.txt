    #[test]
    fn backlog_a_stopped_turn_whose_place_comes_back_before_its_end_is_read() {
        // The runtime sends TurnEnded (worker.rs:3835) before it gives the
        // place back (:3524); the band's settle can run between the two.
        let mut shell = running("stream please");
        shell.apply(UiEvent::Delta("a whole line\n".to_string()));
        shell.flush_paced();
        shell.route_bytes(&[0x03]);
        shell.work.give_back();
        shell.settle_band(Instant::now());
        shell.apply(UiEvent::TurnEnded {
            failure: Some("xfx: the turn was cancelled".to_string()),
        });
        let after = shell.released();
        let text = after.join("\n");
        assert!(!text.contains("system"), "the cancel was reported as a failure: {after:?}");
    }
